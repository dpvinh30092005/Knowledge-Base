function processAIResponse(message) {
const sections = splitResponseIntoSections(message);

let html = '<div class="message-content">';
html += '<i class="fa fa-robot"></i>';

if (sections.mainContent) {
html += '<div class="markdown-content">' + marked.parse(sections.mainContent) + '</div>';
}

if (sections.followUpQuestions && sections.followUpQuestions.length > 0) {
html += '<div class="follow-up-section">';
html += '<div class="follow-up-title">Câu hỏi tiếp theo:</div>';
html += '<div class="follow-up-questions">';

sections.followUpQuestions.forEach((question, index) => {
html += `<div class="follow-up-question" onclick="askAI('${question.replace(/'/g, "\\'")}')">`;
html += `<strong>${index + 1}.</strong> ${question}`;
html += '</div>';
});

html += '</div></div>';
}

if (sections.references && sections.references.length > 0) {
html += '<div class="references-section">';
html += '<div class="references-title">Tài liệu tham khảo:</div>';

sections.references.forEach((ref, index) => {
const url = extractURL(ref);
if (url) {
html += `<a href="${url}" target="_blank" class="reference-link">`;
html += `<i class="fa fa-external-link"></i> ${ref.replace(/\[.*?\]\s*/, '')}`;
html += '</a>';
} else {
html += `<div class="reference-link">${ref}</div>`;
}
});

html += '</div>';
}

html += '</div>';
return html;
}

function splitResponseIntoSections(message) {
const sections = {
mainContent: '',
followUpQuestions: [],
references: []
};

const followUpMatch = message.match(/### Follow-up Questions\s*([\s\S]*?)(?=###|References:|$)/i);
if (followUpMatch) {
const followUpText = followUpMatch[1].trim();
const questions = followUpText.split(/\d+\.\s+/).filter(q => q.trim());
sections.followUpQuestions = questions.map(q => q.trim()).filter(q => q);
}

const referencesMatch = message.match(/References:\s*([\s\S]*?)(?=###|$)/i);
if (referencesMatch) {
const referencesText = referencesMatch[1].trim();
const refs = referencesText.split(/\n/).filter(r => r.trim());
sections.references = refs.map(r => r.trim()).filter(r => r);
}

let mainContent = message;
if (followUpMatch) {
mainContent = mainContent.replace(followUpMatch[0], '');
}
if (referencesMatch) {
mainContent = mainContent.replace(referencesMatch[0], '');
}

sections.mainContent = mainContent.trim();
return sections;
}

function extractURL(text) {
const urlMatch = text.match(/https?:\/\/[^\s]+/);
return urlMatch ? urlMatch[0] : null;
}
